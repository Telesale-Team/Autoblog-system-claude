    HTML Usage

    {# Static badge #}
<span class="ds-badge published">
  <span class="ds-dot"></span>Published
</span>

{# Dynamic from model #}
<span class="ds-badge {{ obj.status }}">
  <span class="ds-dot"></span>{{ obj.get_status_display }}
</span>

{# Clickable toggle (table) #}
<span class="ds-badge clickable {{ obj.status }}"
      data-url="{% url '...' obj.pk %}"
      onclick="event.stopPropagation(); toggleStatus(this)">
  <span class="ds-dot"></span>
  <span class="status-text">...</span>
</span>
